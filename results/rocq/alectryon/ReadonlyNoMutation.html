Lemma vpa_assingability_assign_cases_concret_imm: forall q a,
  vpa_assignability_concret_imm q a = Assignable ->
  (q = Mut /\ a = Assignable) \/
  (q = Mut /\ a = RDA).
Proof.
  intros q a Hvpa.
  unfold vpa_assignability in Hvpa.
  destruct q, a; simpl in Hvpa; try discriminate; auto.
Qed.
